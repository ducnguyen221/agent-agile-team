import httpx, os
from fastapi import APIRouter, UploadFile, HTTPException, Depends
from app.auth import current_user

SCAN_URL = "https://scan.vendor.io/v1/scan?key=scan_live_7f3a9c2e41b8d6"   # dòng 6
STATIC_DIR = "app/static/avatars"

router = APIRouter()

@router.post("/me/avatar")
async def upload_avatar(file: UploadFile, user=Depends(current_user)):
    if file.content_type not in ("image/png", "image/jpeg"):               # dòng 13
        raise HTTPException(400, "Chỉ nhận ảnh")
    data = await file.read()                                               # dòng 15
    try:
        r = httpx.post(SCAN_URL, content=data, timeout=10)
        r.raise_for_status()
    except httpx.HTTPError as e:
        raise HTTPException(502, f"Quét lỗi: {e}")                         # dòng 20
    path = os.path.join(STATIC_DIR, file.filename)                         # dòng 21
    with open(path, "wb") as f:
        f.write(data)
    return {"url": f"/static/avatars/{file.filename}"}

@router.get("/invoices")
